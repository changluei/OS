/* Temporary diagnostic fixture; not part of the submitted kernel. */
#include <sbi.h>

static int fail_pcb, fail_stack;
static size_t pcb_calls, pcb_frees, stack_calls, stack_frees;
void *__real_kmalloc(size_t n);
void __real_kfree(void *p);
struct Page *__real_alloc_pages(size_t n);
void __real_free_pages(struct Page *p, size_t n);

void *__wrap_kmalloc(size_t n) {
    if (n == sizeof(struct proc_struct)) {
        pcb_calls++;
        if (fail_pcb) { fail_pcb = 0; return NULL; }
    }
    return __real_kmalloc(n);
}
void __wrap_kfree(void *p) {
    pcb_frees++;
    __real_kfree(p);
}
struct Page *__wrap_alloc_pages(size_t n) {
    if (n == KSTACKPAGE) {
        stack_calls++;
        if (fail_stack) { fail_stack = 0; return NULL; }
    }
    return __real_alloc_pages(n);
}
void __wrap_free_pages(struct Page *p, size_t n) {
    if (n == KSTACKPAGE) stack_frees++;
    __real_free_pages(p, n);
}

static size_t audit_nodes(list_entry_t *head) {
    size_t n = 0;
    for (list_entry_t *le = list_next(head); le != head; le = list_next(le)) {
        assert(++n <= MAX_PROCESS);
    }
    return n;
}
static size_t audit_hash_nodes(void) {
    size_t n = 0;
    for (size_t i = 0; i < HASH_LIST_SIZE; i++) n += audit_nodes(hash_list + i);
    return n;
}
static void audit_pass(const char *name) { cprintf("[2412819] PASS: %s\n", name); }
static void audit_release(struct proc_struct *p) {
    list_del(&p->list_link);
    list_del(&p->hash_link);
    nr_process--;
    put_kstack(p);
    kfree(p);
}
static void audit_frame(struct proc_struct *p) {
    assert((uintptr_t)p->tf >= p->kstack);
    assert((uintptr_t)p->tf + sizeof(*p->tf) <= p->kstack + KSTACKSIZE);
    assert(((uintptr_t)p->tf & 15) == 0);
    assert(p->context.ra == (uintptr_t)forkret);
    assert(p->context.sp == (uintptr_t)p->tf);
    assert(p->tf->gpr.a0 == 0);
    uintptr_t *words = (uintptr_t *)&p->context;
    for (size_t i = 2; i < sizeof(p->context) / sizeof(*words); i++) assert(words[i] == 0);
}
static int audit_never_called(void *arg) {
    panic("kernel_thread must not directly call the entry function");
}

static void audit_failures(void) {
    struct trapframe tf = {0};
    size_t pages_before = nr_free_pages();
    int count_before = nr_process;
    size_t list_before = audit_nodes(&proc_list), hash_before = audit_hash_nodes();
    size_t ac = pcb_calls, sc = stack_calls, pf = pcb_frees, sf = stack_frees;
    nr_process = MAX_PROCESS;
    assert(do_fork(0, 0, &tf) == -E_NO_FREE_PROC);
    nr_process = count_before;
    assert(pcb_calls == ac && stack_calls == sc);
    audit_pass("process limit rejects before resource allocation");

    fail_pcb = 1;
    assert(do_fork(0, 0, &tf) == -E_NO_MEM && fail_pcb == 0);
    assert(stack_calls == sc && pcb_frees == pf && stack_frees == sf);
    audit_pass("PCB allocation failure returns ENOMEM without registration");

    fail_stack = 1;
    assert(do_fork(0, 0, &tf) == -E_NO_MEM && fail_stack == 0);
    assert(pcb_frees == pf + 1 && stack_frees == sf);
    audit_pass("kernel-stack allocation failure releases PCB");

    struct mm_struct dummy_mm;
    current->mm = &dummy_mm;
    assert(do_fork(CLONE_VM, 0, &tf) == -E_INVAL);
    current->mm = NULL;
    assert(pcb_frees == pf + 2 && stack_frees == sf + 1);
    assert(nr_free_pages() == pages_before);
    assert(nr_process == count_before);
    assert(audit_nodes(&proc_list) == list_before && audit_hash_nodes() == hash_before);
    audit_pass("unsupported user mm cleans stack and PCB, preserves all registries");
}

static void audit_copy(void) {
    struct trapframe source, saved;
    memset(&source, 0x5a, sizeof(source));
    saved = source;
    struct proc_struct *p = alloc_proc();
    assert(p != NULL && setup_kstack(p) == 0);
    memset(&p->context, 0xa5, sizeof(p->context));
    copy_thread(p, 0, &source);
    audit_frame(p);
    assert(p->tf != &source && p->tf->gpr.sp == (uintptr_t)p->tf);
    struct trapframe expected = source;
    expected.gpr.a0 = 0;
    expected.gpr.sp = (uintptr_t)p->tf;
    assert(memcmp(p->tf, &expected, sizeof(expected)) == 0);
    assert(memcmp(&source, &saved, sizeof(source)) == 0);
    audit_pass("copy_thread copies the full frame, clears context, isolates parent stack");

    uintptr_t explicit_sp = (p->kstack + KSTACKSIZE - 512) & ~(uintptr_t)15;
    copy_thread(p, explicit_sp, &source);
    audit_frame(p);
    assert(p->tf->gpr.sp == explicit_sp);
    assert(memcmp(&source, &saved, sizeof(source)) == 0);
    put_kstack(p);
    kfree(p);
    audit_pass("copy_thread preserves a supplied aligned stack pointer");
}

static void audit_success(void) {
    struct proc_struct *parent = current;
    size_t pages_before = nr_free_pages();
    size_t list_before = audit_nodes(&proc_list), hash_before = audit_hash_nodes();
    int count_before = nr_process;
    uintptr_t status = read_csr(sstatus);
    assert(!(status & SSTATUS_SIE));
    int pid1 = kernel_thread(audit_never_called, NULL, CLONE_THREAD);
    struct proc_struct *a = find_proc(pid1);
    assert(pid1 > 0 && a != NULL && a != parent && a != initproc);
    assert(current == parent && a->parent == parent);
    assert(a->mm == NULL && a->pgdir == boot_pgdir_pa && a->state == PROC_RUNNABLE);
    audit_frame(a);
    assert(a->tf->epc == (uintptr_t)kernel_thread_entry);
    assert(a->tf->gpr.s0 == (uintptr_t)audit_never_called && a->tf->gpr.s1 == 0);
    assert(a->tf->status == ((status | SSTATUS_SPP | SSTATUS_SPIE) & ~SSTATUS_SIE));
    assert(!(read_csr(sstatus) & SSTATUS_SIE));
    audit_pass("kernel_thread builds entry, null argument and S-mode status without running it");

    intr_enable();
    status = read_csr(sstatus);
    int pid2 = kernel_thread(audit_never_called, (void *)0x2412819, 0);
    assert(read_csr(sstatus) & SSTATUS_SIE);
    intr_disable();
    struct proc_struct *b = find_proc(pid2);
    assert(pid2 > 0 && pid2 != pid1 && b != NULL && b != a);
    assert(current == parent && b->parent == parent && b->state == PROC_RUNNABLE);
    assert(b->tf->gpr.s1 == 0x2412819);
    assert(b->tf->status == ((status | SSTATUS_SPP | SSTATUS_SPIE) & ~SSTATUS_SIE));
    assert(a->kstack + KSTACKSIZE <= b->kstack || b->kstack + KSTACKSIZE <= a->kstack);
    audit_frame(b);
    assert(nr_process == count_before + 2);
    assert(audit_nodes(&proc_list) == list_before + 2 && audit_hash_nodes() == hash_before + 2);
    audit_pass("unique PIDs, parent, independent stacks, registry counts and interrupt restoration");
    audit_release(a);
    audit_release(b);
    assert(find_proc(pid1) == NULL && find_proc(pid2) == NULL);
    assert(nr_process == count_before && nr_free_pages() == pages_before);
    assert(audit_nodes(&proc_list) == list_before && audit_hash_nodes() == hash_before);
    audit_pass("diagnostic cleanup restores allocated stack pages and process registries");
}

static int audit_entry(void *arg) {
    assert(arg == (void *)0x2412819);
    assert(current != idleproc && current->state == PROC_RUNNABLE);
    uintptr_t sp;
    asm volatile("mv %0, sp" : "=r"(sp));
    assert(sp >= current->kstack && sp < current->kstack + KSTACKSIZE);
    assert((sp & 15) == 0);
    assert((read_csr(sstatus) & SSTATUS_SPP) == 0);
    assert(read_csr(sstatus) & SSTATUS_SIE);
    audit_pass("real forkret/forkrets/sret/kernel_thread_entry reaches fn(arg) on child stack");
    cprintf("[2412819] ALL CREATION TESTS PASSED\n");
    sbi_shutdown();
    while (1) ;
}

/* Deliberately no scheduler/proc_run/switch_to implementation: test only creation. */
void cpu_idle(void) {
    intr_disable();
    assert(current == idleproc && initproc != NULL && initproc->pid == 1);
    assert(initproc->parent == idleproc && initproc->mm == NULL);
    assert(initproc->state == PROC_RUNNABLE && initproc->pgdir == boot_pgdir_pa);
    audit_frame(initproc);
    audit_pass("original proc_init creates and registers initproc with the real four functions");
    audit_copy();
    audit_failures();
    audit_success();
    int pid = kernel_thread(audit_entry, (void *)0x2412819, 0);
    assert(pid > 0 && find_proc(pid) != NULL);
    current = find_proc(pid);
    forkret();
    panic("forkret unexpectedly returned");
}
