五、针对性模拟与汇编
Ptolemy 创建 /tmp/lab4-function3-test 夹具，直接包含正式 sched.c，从正式 proc.c 提取本人 function3；主 agent 重新执行 /tmp/lab4-function3-test/test，结果：
PASS: 2562 scheduler cases; proc_run same-target/direct switch; Sv39 high PA, TLB ordering and nested SIE restoration (mock switch).
边界：switch_to、CSR 与 flush_tlb 均为替身，用于检查调度选择、状态、调用顺序与恢复，不证明硬件寄存器保存或不同地址空间。正式 QEMU 课程测试补充真实首次运行及线程往返。夹具不增加到正式代码树，下面保留其源文件便于复核。

--- 模拟夹具 test.c ---
#include <proc.h>
#include <sync.h>
#include <sched.h>
#include <assert.h>

extern int puts(const char *text);
list_entry_t proc_list;
struct proc_struct *current, *idleproc, *initproc;
unsigned long test_sstatus;
static unsigned long long test_satp;
static int writes, flushes, switches;
static struct proc_struct *selected;
static struct proc_struct idle, threads[4];

void intr_disable(void) { test_sstatus &= ~SSTATUS_SIE; }
void intr_enable(void) { test_sstatus |= SSTATUS_SIE; }
void test_write_satp(unsigned long long value) {
    assert(!(test_sstatus & SSTATUS_SIE));
    assert(value == (0x8000000000000000ULL | (current->pgdir >> 12)));
    test_satp = value;
    writes++;
}
static void flush_tlb(void) {
    assert(!(test_sstatus & SSTATUS_SIE));
    assert(writes == flushes + 1);
    flushes++;
}
void switch_to(struct context *from, struct context *to) {
    assert(!(test_sstatus & SSTATUS_SIE));
    assert(writes == flushes && flushes == switches + 1);
    assert(to == &current->context);
    assert(test_satp == (0x8000000000000000ULL | (current->pgdir >> 12)));
    selected = current;
    switches++;
    current = to_struct(from, struct proc_struct, context);
    test_satp = 0x8000000000000000ULL | (current->pgdir >> 12);
}
#include "proc-functions.inc"
#include "/home/ljz/course/os/OS/code/kern/schedule/sched.c"

static void reset(int count) {
    list_init(&proc_list);
    idle = (struct proc_struct){ .state = PROC_RUNNABLE, .pgdir = 0x80000000ULL };
    idleproc = &idle;
    for (int index = 0; index < count; index++) {
        threads[index] = (struct proc_struct){ .pgdir = 0x123456789000ULL + (index << 12) };
        list_add_before(&proc_list, &threads[index].list_link);
    }
    writes = flushes = switches = 0;
    selected = NULL;
}
int main(void) {
    int cases = 0;
    for (int enabled = 0; enabled < 2; enabled++) {
        reset(0);
        current = idleproc;
        current->need_resched = 1;
        test_sstatus = enabled ? SSTATUS_SIE : 0;
        schedule();
        assert(current == idleproc && !current->need_resched && idle.runs == 1);
        assert(!switches && test_sstatus == (enabled ? SSTATUS_SIE : 0));
        cases++;
        for (int states = 0; states < 256; states++) {
            for (int origin = -1; origin < 4; origin++) {
                reset(4);
                for (int index = 0; index < 4; index++) {
                    threads[index].state = (states >> (2 * index)) & 3;
                    threads[index].need_resched = 1;
                }
                idle.need_resched = 1;
                struct proc_struct *source = origin < 0 ? idleproc : &threads[origin];
                struct proc_struct *expected = idleproc;
                for (int offset = 1; offset <= 4; offset++) {
                    int index = (origin + offset) % 4;
                    if (threads[index].state == PROC_RUNNABLE) {
                        expected = &threads[index];
                        break;
                    }
                }
                current = source;
                test_sstatus = enabled ? SSTATUS_SIE : 0;
                schedule();
                assert(current == source && !source->need_resched);
                assert(expected->runs == 1);
                assert(switches == (source != expected));
                assert(source == expected || selected == expected);
                assert(test_sstatus == (enabled ? SSTATUS_SIE : 0));
                for (int index = 0; index < 4; index++) {
                    assert(threads[index].state == ((states >> (2 * index)) & 3));
                    assert(threads[index].runs == (&threads[index] == expected));
                    assert(list_next(index ? &threads[index - 1].list_link : &proc_list) == &threads[index].list_link);
                    assert(threads[index].need_resched == (&threads[index] != source));
                }
                cases++;
            }
        }
        reset(1);
        current = idleproc;
        test_sstatus = enabled ? SSTATUS_SIE : 0;
        proc_run(current);
        assert(!writes && !flushes && !switches && current == idleproc);
        threads[0].state = PROC_RUNNABLE;
        proc_run(&threads[0]);
        assert(switches == 1 && selected == &threads[0] && current == idleproc);
        assert(!threads[0].runs && test_sstatus == (enabled ? SSTATUS_SIE : 0));
    }
    assert(cases == 2562);
    puts("PASS: 2562 scheduler cases; proc_run same-target/direct switch; Sv39 high PA, TLB ordering and nested SIE restoration (mock switch).");
    return 0;
}

//LAB4 function3 2412852
void cpu_idle(void)
{
    while (1)
    {
        if (current->need_resched)
        {
            schedule();
        }
    }
}

void proc_run(struct proc_struct *proc)
{
    if (proc != current)
    {
        bool intr_flag;
        local_intr_save(intr_flag);
        struct proc_struct *prev = current;
        current = proc;
        write_csr(satp, 0x8000000000000000ULL | (proc->pgdir >> RISCV_PGSHIFT));
        flush_tlb();
        switch_to(&prev->context, &proc->context);
        local_intr_restore(intr_flag);
    }
}
